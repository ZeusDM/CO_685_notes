\lecture{2}{2026-09-15}{Extended Euclidean Algorithm}

\begin{definition}
    The \emph{ring $\mathbb{Z}_n$ of integers modulo $n$} is the set $\{0, 1, \ldots, n-1\}$ with addition and multiplication defined modulo $n$.
    The \emph{multiplicative group of units modulo $n$}, denoted $\mathbb{Z}_n^*$, is the set of elements in $\mathbb{Z}_n$ that have a multiplicative inverse modulo $n$.
\end{definition}

\begin{theorem}
    For any integer \(n \geq 1\), \(\mathbb{Z}_n^* = \{a \in \mathbb{Z}_n : \gcd(a, n) = 1\}\).
\end{theorem}

\begin{proof}
    If \(a \in \mathbb{Z}_n^*\), there exists \(b \in \mathbb{Z}_n\) with \(ab \equiv 1 \pmod n\), so \(ab - kn = 1\) for some \(k \in \mathbb{Z}\). Any common divisor of \(a\) and \(n\) divides \(ab - kn = 1\), so \(\gcd(a, n) = 1\).
    If \(\gcd(a, n) = 1\), by the Extended Euclidean Algorithm, there exist \(x, y \in \mathbb{Z}\) such that \(ax + ny = 1\). Modulo \(n\), \(ax \equiv 1 \pmod n\), so \(a \pmod n \in \mathbb{Z}_n^*\).
\end{proof}

\section*{Extended Euclidean Algorithm}

\begin{algorithm}
    \caption{Extended Euclidean Algorithm}
    \begin{algorithmic}[1]
        \Require Integers \(n \geq m \geq 0\)
        \Ensure Integers \(a, b\) such that \(an + bm = \gcd(m, n)\)
        \State initialize \(r_0 = n\) and \(r_1 = m\)
        \State initialize \(a_0 = 1\) and \(a_1 = 0\)
        \State initialize \(b_0 = 0\) and \(b_1 = 1\)
        \State initialize \(i \gets 1\)
        \While{\(r_i \neq 0\)}
            \State set \(q_i\) and \(r_{i+1}\) such that \(r_{i-1} = q_i r_i + r_{i+1}\) via integer division
            \State set \(a_{i+1} = a_{i-1} - q_i a_i\)
            \State set \(b_{i+1} = b_{i-1} - q_i b_i\)
            \State \(i \gets i + 1\)
        \EndWhile
        \State \Return \((a, b) = (a_{i-1}, b_{i-1})\)
    \end{algorithmic}
\end{algorithm}

\begin{theorem}
    The Extended Euclidean Algorithm terminates at some step \(k\) where \(r_{k+1} = 0\), yielding \(r_k = \gcd(m, n) = a_k n + b_k m\).
\end{theorem}

\begin{proof}
    First, we claim that \(a_i n + b_i m = r_i\) for all suitable \(i\).
    We verify this by induction.
    If \(i \in \{0, 1\}\), the result is immediate.
    If \(i > 1\), assuming the claim for \(i-1\) and \(i\),
    \[
        a_{i+1} n + b_{i+1} m = (a_{i-1} n + b_{i-1} m) - q_i (a_i n + b_i m) = r_{i-1} - q_i r_i = r_{i+1}.
    \]

    Next, we claim that \(\gcd(r_{i-1}, r_i) = \gcd(r_i, r_{i+1})\) for all suitable \(i\).
    Because \(r_{i+1} = r_{i-1} - q_i r_i\) and \(r_{i-1} = q_i r_i + r_{i+1}\), an integer \(d\) divides both \(r_{i-1}\) and \(r_i\) if and only if \(d\) divides both \(r_i\) and \(r_{i+1}\). Hence the set of common divisors of \((r_{i-1}, r_i)\) equals that of \((r_i, r_{i+1})\), so their greatest common divisors are equal.

    Finally, the remainders form a strictly decreasing sequence of non-negative integers \(r_0 > r_1 > \dots \geq 0\), so the algorithm must terminate with \(r_{k+1} = 0\).
    Then \(\gcd(m, n) = \gcd(r_0, r_1) = \dots = \gcd(r_k, r_{k+1}) = \gcd(r_k, 0) = r_k\), and \(a_k n + b_k m = r_k = \gcd(m, n)\).
\end{proof}

\begin{theorem}
    Let \(k\) be the index at which the Extended Euclidean Algorithm terminates with \(r_{k+1} = 0\). Then \(k \leq 2 \lceil \log_2 n \rceil + 1\).
\end{theorem}

\begin{proof}
    First, we claim that for all \(i \geq 1\) with \(r_i \neq 0\), \(r_{i+1} < \frac{1}{2} r_{i-1}\).
    If \(q_i \geq 2\), then \(r_{i-1} \geq 2 r_i > 2 r_{i+1}\), so \(r_{i+1} < \frac{1}{2} r_{i-1}\).
    If \(q_i = 1\), then \(r_{i-1} = r_i + r_{i+1} > 2 r_{i+1}\), so \(r_{i+1} < \frac{1}{2} r_{i-1}\).

    Applying this inequality repeatedly yields \(r_{2j+1} < \frac{1}{2^j} r_1 \leq \frac{n}{2^j}\). For \(j > \log_2 n\), we have \(r_{2j+1} < 1\), which implies \(r_{2j+1} = 0\). Thus, \(k \leq 2 \lceil \log_2 n \rceil + 1\).
\end{proof}

\subsection*{Runtime}

Because we work with large integers in this course, we care about runtimes in terms of the bit-length of the integers.
Let \(a, b, n\) be integers such that \(a, b \leq n\), and \(2^{r-1} \leq n < 2^r\), so that the operands have bit-length at most \(r\).
The bit-complexity of standard arithmetic operations is summarized in \Cref{tab:arithmetic_runtimes}.

\begin{table}[h]
    \centering
    \begin{tabular}{lc}
        \toprule
        Operation & Runtime \\
        \midrule
        Adding: \(a + b\) & \(O(r)\) \\
        Subtraction: \(a - b\) & \(O(r)\) \\
        Comparison: \(a \overset{?}{<} b\) & \(O(r)\) \\
        Multiplication: \(ab\) & \(O(r^2)\) \\
        Integer Division: \(a = bq + r\) & \(O(r^2)\) \\
        \midrule
        Modular Addition or Subtraction: \(a \pm b \pmod n\) & \(O(r)\) \\
        Modular Multiplication: \(ab \pmod n\) & \(O(r^2)\) \\
        Modular Inverse (EEA): \(a^{-1} \pmod n\) & \(O(r^2)\) \\
        Modular Division: \(a^{-1} b \pmod n\) & \(O(r^2)\) \\
        Modular Exponentiation: \(a^b \pmod n\) & \(O(r^3)\) \\
        \bottomrule
    \end{tabular}
    \caption{Modular arithmetic runtimes, where \(a, b \leq n\) and \(2^{r-1} \leq n < 2^r\).}
    \label{tab:arithmetic_runtimes}
\end{table}

Note that for some of these operations, there exist better algorithms strictly asymptotically speaking (such as \(O(r \log r)\) for multiplication and division), but their improvements are only relevant for inputs much larger than the ones we encounter in this course.

From \Cref{tab:arithmetic_runtimes} and a naive analysis of the Extended Euclidean Algorithm, we have \(O(r)\) iterations, and each iteration requires \(O(r^2)\) time for integer division. Thus the overall runtime is \(O(r^3)\).
However, a better analysis shows that the overall runtime is actually \(O(r^2)\) \cite{Collins74}; the key idea is that the numbers get smaller as the algorithm progresses, so the later iterations are faster than the earlier ones.
