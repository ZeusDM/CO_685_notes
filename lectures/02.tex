\lecture{2}{2026-09-15}{Modular Integers}

\begin{definition}
    The \emph{ring of integers modulo $n$}, denoted $\mathbb{Z}_n$, is the set $\{0, 1, 2, \ldots, n-1\}$ with addition and multiplication defined modulo $n$.
    The \emph{multiplicative group of units modulo $n$}, denoted $\mathbb{Z}_n^*$, is the set of elements in $\mathbb{Z}_n$ that have a multiplicative inverse modulo $n$.
\end{definition}

\begin{theorem}
    \(\mathbb{Z}_n^* = \{a \in \mathbb{Z}_n : \gcd(a, n) = 1\}\).
\end{theorem}

\begin{proof}
    If \(a \in \mathbb{Z}_n^*\), there exists \(b \in \mathbb{Z}_n\) with \(ab \equiv 1 \pmod n\), so \(ab - kn = 1\) for some \(k \in \mathbb{Z}\). Any common divisor of \(a\) and \(n\) divides \(ab - kn = 1\), so \(\gcd(a, n) = 1\).
    If \(\gcd(a, n) = 1\), by the Extended Euclidean Algorithm, there exist \(x, y \in \mathbb{Z}\) such that \(ax + ny = 1\). Modulo \(n\), \(ax \equiv 1 \pmod n\), so \(a \pmod n \in \mathbb{Z}_n^*\).
\end{proof}

\begin{algorithm}
    \caption{Extended Euclidean Algorithm}
    \begin{algorithmic}[1]
        \Require Integers \(n \geq x \geq 0\)
        \Ensure Integers \(a, b\) such that \(an + bx = \gcd(x, n)\)
        \State initialize \(r_0 = n\) and \(r_1 = x\)
        \State initialize \(a_0 = 1\) and \(a_1 = 0\)
        \State initialize \(b_0 = 0\) and \(b_1 = 1\)
        \State initialize \(i \gets 1\)
        \While{\(r_i \neq 0\)}
            \State set \(q_i\) and \(r_{i+1}\) such that \(r_{i-1} = q_i r_i + r_{i+1}\) and \(0 \leq r_{i+1} < r_i\) from integer division
            \State set \(a_{i+1} = a_{i-1} - q_i a_i\)
            \State set \(b_{i+1} = b_{i-1} - q_i b_i\)
            \State \(i \gets i + 1\)
        \EndWhile
    \end{algorithmic}
\end{algorithm}

\begin{lemma}[B\'ezout Invariant]
    For all \(i \geq 0\), \(a_i n + b_i x = r_i\).
\end{lemma}

\begin{proof}
    For \(i = 0\), \(1 \cdot n + 0 \cdot x = r_0\); for \(i = 1\), \(0 \cdot n + 1 \cdot x = r_1\). Assuming the claim for \(i-1\) and \(i\),
    \(
        a_{i+1} n + b_{i+1} x = (a_{i-1} n + b_{i-1} x) - q_i (a_i n + b_i x) = r_{i-1} - q_i r_i = r_{i+1}
    \). \qedhere
\end{proof}

\begin{lemma}
    For all \(i \geq 1\) with \(r_i \neq 0\), \(\gcd(r_{i-1}, r_i) = \gcd(r_i, r_{i+1})\).
\end{lemma}

\begin{proof}
    Because \(r_{i+1} = r_{i-1} - q_i r_i\) and \(r_{i-1} = q_i r_i + r_{i+1}\), an integer \(d\) divides both \(r_{i-1}\) and \(r_i\) if and only if \(d\) divides both \(r_i\) and \(r_{i+1}\). Hence the set of common divisors of \((r_{i-1}, r_i)\) equals that of \((r_i, r_{i+1})\), so their greatest common divisors are equal.
\end{proof}

\begin{theorem}
    The algorithm terminates at some step \(k\) where \(r_{k+1} = 0\), yielding \(r_k = \gcd(x, n) = a_k n + b_k x\).
\end{theorem}

\begin{proof}
    The remainders form a strictly decreasing sequence of non-negative integers \(r_0 > r_1 > \dots \geq 0\), so the algorithm must terminate with \(r_{k+1} = 0\). By Lemma 2.3, \(\gcd(x, n) = \gcd(r_0, r_1) = \dots = \gcd(r_k, r_{k+1}) = \gcd(r_k, 0) = r_k\). By Lemma 2.2, \(a_k n + b_k x = r_k = \gcd(x, n)\).
\end{proof}

\begin{lemma}[Halving]
    For all \(i \geq 1\) with \(r_i \neq 0\), \(r_{i+1} < \frac{1}{2} r_{i-1}\).
\end{lemma}

\begin{proof}
    Since \(0 \leq r_{i+1} < r_i\).
    If \(r_i \leq \frac{1}{2} r_{i-1}\), then \(r_{i+1} < r_i \leq \frac{1}{2} r_{i-1}\).
    If \(r_i > \frac{1}{2} r_{i-1}\), then \(q_i = 1\), so \(r_{i+1} = r_{i-1} - r_i < r_{i-1} - \frac{1}{2} r_{i-1} = \frac{1}{2} r_{i-1}\). \qedhere
\end{proof}

\begin{theorem}[Runtime]
    The Extended Euclidean Algorithm runs in \(O(\log n)\) iterations.
\end{theorem}

\begin{proof}
    Applying the Halving Lemma repeatedly gives \(r_{2j+1} < \frac{1}{2^j} r_1 \leq \frac{n}{2^j}\). For \(j > \log_2 n\), we have \(r_{2j+1} < 1\), which implies \(r_{2j+1} = 0\). Thus the algorithm terminates in at most \(2 \lceil \log_2 n \rceil + 1 = O(\log n)\) iterations.
\end{proof}

From a naive analysis of the Extended Euclidean Algorithm, we have \(O(\log n)\) iterations, and each iteration requires \(O((\log n)^2)\) time for integer division. Thus the overall runtime is \(O((\log n)^3)\).
However, a better analysis of the same algorithm shows that the overall runtime is \(O((\log n)^2)\).

\section*{Primitive Roots}

\begin{definition}[Primitive Root]
    A \emph{generator} \(g\) of a group \(G\) is an element such that every element of \(G\) can be expressed as a power of \(g\).
    The generators of the multiplicative group \(\mathbb{Z}_n^*\) are called \emph{primitive roots of \(n\)}.
\end{definition}

\begin{theorem}
    If \(n\) is prime, then there exists a primitive root modulo \(n\).
\end{theorem}

\begin{proof}
    For each divisor \(d \mid (n-1)\), let \(\psi(d)\) be the number of elements in \(\mathbb{Z}_n^*\) of order \(d\).
    If \(\psi(d) > 0\), fix an element \(h\) of order \(d\). The \(d\) distinct powers \(1, h, \dots, h^{d-1}\) satisfy \(x^d = 1\). Since \(x^d - 1\) has at most \(d\) roots over the field \(\mathbb{Z}_n\), these powers comprise all solutions.
    Among them, \(h^k\) has order \(d\) if and only if \(\gcd(k, d) = 1\), of which there are \(\phi(d)\) choices.
    Hence, \(\psi(d) = 0\) or \(\psi(d) = \phi(d)\); in either case, \(\psi(d) \leq \phi(d)\) for all \(d\).

    Since each of the \(n-1\) elements of \(\mathbb{Z}_n^*\) has order dividing \(n-1\), and \(\sum_{d \mid n-1} \phi(d) = n-1\),
    \[
        n - 1 = \sum_{d \mid n-1} \psi(d) \leq \sum_{d \mid n-1} \phi(d) = n - 1.
    \]
    Thus \(\psi(d) = \phi(d)\) for all \(d \mid (n-1)\). In particular, \(\psi(n-1) = \phi(n-1) \geq 1\), so a primitive root \(g\) (of order \(n-1\)) exists.
\end{proof}
