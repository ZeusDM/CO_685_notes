\lecture{4}{2026-09-22}{The Legendre Symbol and Modular Roots}

\begin{definition}
    Let \(p\) be an odd prime.
    The \emph{Legendre symbol} is defined for \(a \in \mathbb{Z}\) by
    \[
        \legendre{a}{p} =
        \begin{cases}
            0 & \text{if } p \mid a, \\
            1 & \text{if } a \in \operatorname{QR}_p, \\
            -1 & \text{if } a \in \overline{\operatorname{QR}}_p.
        \end{cases}
    \]
\end{definition}

\begin{theorem}[Euler's Criterion] \label{thm:euler-criterion}
    Let \(p\) be an odd prime and \(a \in \mathbb{Z}\) with \(p \nmid a\).
    Then
    \[
        \legendre{a}{p} \equiv a^{\frac{p-1}{2}} \pmod{p}.
    \]
\end{theorem}

We prove a more general result about subgroups of \(\mathbb{Z}_p^\times\) first.

\begin{theorem} \label{thm:unique-subgroup-of-order-d}
    Let \(p > 2\) be prime and \(d \mid (p-1)\). Then:
    \begin{enumerate}[label=\textup{(\roman*)}]
        \item \label{itm:unique-subgroup}
        \(\mathbb{Z}_p^\times\) has a unique subgroup of order \(d\).
        \item \label{itm:generator-of-subgroup}
        For any generator \(g\) of \(\mathbb{Z}_p^\times\), \(g^{\frac{p-1}{d}}\) generates this unique subgroup.
        \item \label{itm:surjective-homomorphism}
        The map \(x \mapsto x^{\frac{p-1}{d}}\) is a surjective homomorphism from \(\mathbb{Z}_p^\times\) onto this subgroup, with kernel being the unique subgroup of order \(\frac{p-1}{d}\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    Since \(\mathbb{Z}_p^\times = \langle g \rangle\) is cyclic of order \(p-1\), \(g^{\frac{p-1}{d}}\) has order \(d\).
    Because \(x^d - 1\) has at most \(d\) roots in \(\mathbb{Z}_p\), \(\langle g^{\frac{p-1}{d}} \rangle\) is the unique subgroup of order \(d\), proving \ref{itm:unique-subgroup} and \ref{itm:generator-of-subgroup}.
    For \ref{itm:surjective-homomorphism}, the power map \(\phi \colon x \mapsto x^{\frac{p-1}{d}}\) is a homomorphism with \(\phi(g^k) = (g^{\frac{p-1}{d}})^k\), so \(\operatorname{im}(\phi) = \langle g^{\frac{p-1}{d}} \rangle\); its kernel consists of all roots of \(x^{\frac{p-1}{d}} \equiv 1 \pmod{p}\), which is the unique subgroup of order \(\frac{p-1}{d}\).
\end{proof}

\begin{proof}[Proof of \nameref{thm:euler-criterion}]
    If \(p \mid a\), then both sides are \(0\).
    Otherwise, let \(g\) generate \(\mathbb{Z}_p^\times\), and write \(a \equiv g^{2k + b} \pmod{p}\) for \(k \in \mathbb{Z}\) and \(b \in \{0, 1\}\).
    Then, \(a\) is a quadratic residue if and only if \(b = 0\), so \(\legendre{a}{p} = (-1)^b\).
    By \Cref{thm:unique-subgroup-of-order-d}\ref{itm:generator-of-subgroup} with \(d = 2\), \(g^{\frac{p-1}{2}}\) generates the unique subgroup of order \(2\) in \(\mathbb{Z}_p^\times\), so \(g^{\frac{p-1}{2}} \equiv -1 \pmod{p}\).
    Then
    \[
        a^{\frac{p-1}{2}} \equiv (g^{2k + b})^{\frac{p-1}{2}} = g^{k(p-1)} (g^{\frac{p-1}{2}})^b \equiv (g^{\frac{p-1}{2}})^b \equiv (-1)^b \equiv \legendre{a}{p} \pmod{p}. \qedhere
    \]
\end{proof}

\begin{corollary}
    Let \(p\) be an odd prime and \(a, b \in \mathbb{Z}\). Then
    \[
        \legendre{ab}{p} = \legendre{a}{p} \legendre{b}{p}.
    \]
\end{corollary}

\begin{proof}
    By \nameref{thm:euler-criterion},
    \[
        \legendre{ab}{p} \equiv (ab)^{\frac{p-1}{2}} \equiv a^{\frac{p-1}{2}} b^{\frac{p-1}{2}} \equiv \legendre{a}{p} \legendre{b}{p} \pmod{p},
    \]
    and equality holds since both sides lie in \(\{-1, 0, 1\}\) and \(p > 2\).
\end{proof}

\begin{proposition}
    For an odd prime \(p\),
    \[
        \legendre{-1}{p} = (-1)^{\frac{p-1}{2}} =
        \begin{cases}
            1 & \text{if } p \equiv 1 \pmod{4}, \\
            -1 & \text{if } p \equiv 3 \pmod{4}.
        \end{cases}
    \]
\end{proposition}

\begin{proof}
    By \nameref{thm:euler-criterion}, \(\legendre{-1}{p} \equiv (-1)^{\frac{p-1}{2}} \pmod{p}\), and equality holds since both sides are in \(\{-1, 1\}\).
    The exponent \(\frac{p-1}{2}\) is even if and only if \(p \equiv 1 \pmod{4}\).
\end{proof}

\section*{Modular Roots}

Given \(a \in \mathbb{Z}_p^\times\) and \(k \geq 2\), we wish to find \(b \in \mathbb{Z}_p^\times\) such that \(b^k \equiv a \pmod{p}\).

\subsection*{The Coprime Case: \texorpdfstring{$\gcd(k, p-1) = 1$}{gcd(k, p-1) = 1}}

When \(\gcd(k, p-1) = 1\), the map \(x \mapsto x^k\) is an automorphism of \(\mathbb{Z}_p^\times\).
Compute \(d \equiv k^{-1} \pmod{p-1}\) via the Extended Euclidean Algorithm.
Then \(b \equiv a^d \pmod{p}\) is the unique solution since \(kd = 1 + m(p-1)\) implies
\[
    b^k \equiv a^{kd} = a^{1 + m(p-1)} \equiv a \pmod{p}.
\]

\subsection*{Modular Square Roots: \texorpdfstring{$k = 2$}{k = 2}}

For \(k = 2\), solutions to \(b^2 \equiv a \pmod{p}\) exist if and only if \(a \in \operatorname{QR}_p\) if and only if \(1 = \legendre{a}{p} \equiv a^{\frac{p-1}{2}} \pmod{p}\) by \nameref{thm:euler-criterion}.

When \(p \equiv 3 \pmod{4}\), there is an immediate closed-form solution: \(b \equiv a^{\frac{p+1}{4}} \pmod{p}\) is a square root because
\[
    b^2 \equiv a^{\frac{p+1}{2}} = a^{\frac{p-1}{2}} a \equiv 1 \cdot a \equiv a \pmod{p}.
\]

More generally, for any odd prime \(p\), we can use the \emph{\nameref{alg:modular-square-root}}.
The moral of the algorithm is as follows.
Write \(p - 1 = 2^e r\) with \(r\) odd and set an initial candidate \(b \gets a^{(r+1)/2}\).
Then \(b^2a^{-1}\) has order \(2^i\) for some \(i \leq e - 1\) since
\[
    (b^2 a^{-1})^{2^{e-1}} = (a^{r+1} a^{-1})^{2^{e-1}} = a^{r 2^{e-1}} = a^{\frac{p-1}{2}} \equiv 1 \pmod{p}.
\]
Choose a quadratic non-residue \(x\) and set \(c = x^r\), which has order \(2^e\).
Repeatedly multiply \(b\) by an appropriate power of \(c\) to decrease the order of \(b^2a^{-1}\) until it becomes \(1\).

\begin{algorithm}[htbp]
    \caption{Tonelli--Shanks Algorithm}\label{alg:modular-square-root}
    \begin{algorithmic}[1]
        \Require odd prime \(p\), \(a \in \operatorname{QR}_p\), and \(x \in \overline{\operatorname{QR}}_p\)
        \Ensure \(b \in \mathbb{Z}_p^\times\) such that \(b^2 \equiv a \pmod{p}\)
        \State factor \(p - 1 = 2^e r\) with \(r\) odd
        \State set \(c = x^r \pmod{p}\) \Comment{\(\operatorname{ord}(c) = 2^e\)}
        \State set \(b \gets a^{\frac{r+1}{2}} \pmod{p}\) \Comment{\(\operatorname{ord}(b^2 a^{-1}) \mid 2^{e-1}\)}
        \While{\(b^2 a^{-1} \not\equiv 1 \pmod{p}\)} \label{alg:modular-square-root:outer-loop}
            \State set \(i \gets 1\) and \(t \gets (b^2 a^{-1})^2 \pmod{p}\)
            \While{\(t \not\equiv 1 \pmod{p}\)} \label{alg:modular-square-root:inner-loop}
                \State increment \(i \gets i + 1\) and square \(t \gets t^2 \pmod{p}\)
            \EndWhile \Comment{\(2^i = \operatorname{ord}(b^2 a^{-1})\)}
            \State set \(b \gets b \cdot c^{2^{e-i-1}} \pmod{p}\) \label{alg:modular-square-root:update-b} \Comment{\(\operatorname{ord}(b^2 a^{-1}) \leq 2^{i-1}\)}
        \EndWhile
    \end{algorithmic}
\end{algorithm}

\begin{remark}
    The required quadratic non-residue \(x \in \overline{\operatorname{QR}}_p\) is relatively straightforward to find in expected \(O(\log^3 p)\) time: since \(|\overline{\operatorname{QR}}_p| = \frac{p-1}{2}\), a uniformly sampled element is a non-residue with probability \(1/2\), requiring an expected \(2 = O(1)\) trials, each tested in \(O(\log^3 p)\) time via Euler's criterion.
\end{remark}

\begin{theorem}
    The \nameref{alg:modular-square-root} terminates and works.
\end{theorem}

\begin{proof}
    Initially, \((b^2 a^{-1})^{2^{e-1}} \equiv (a^r)^{2^{e-1}} = a^{\frac{p-1}{2}} \equiv 1 \pmod{p}\), so \(\operatorname{ord}(b^2 a^{-1})\) is a power of \(2\) dividing \(2^{e-1}\).

    Assume that when the outer while loop in line~\ref{alg:modular-square-root:outer-loop} is reached, \(\operatorname{ord}(b^2 a^{-1})\) is a power of \(2\); which we checked above for its first iteration.
    Then, the inner while loop in line~\ref{alg:modular-square-root:inner-loop} guarantees that \(2^i = \operatorname{ord}(b^2 a^{-1})\) when it terminates.

    Updating \(b' = b \cdot c^{2^{e-i-1}}\) in line~\ref{alg:modular-square-root:update-b}, both \((b^2 a^{-1})^{2^{i-1}}\) and \(c^{2^{e-1}}\) have order \(2\) in \(\mathbb{Z}_p^\times\), so both equal \(-1\). Thus,
    \[
        ((b')^2 a^{-1})^{2^{i-1}} = (b^2 a^{-1})^{2^{i-1}} c^{2^{e-1}} \equiv (-1)(-1) = 1 \pmod{p},
    \]
    which implies that \(\operatorname{ord}((b')^2 a^{-1})\) divides \(2^{i-1}\).
    Hence, each iteration of the outer while loop guarantees that \(\log_2 \operatorname{ord}(b^2 a^{-1})\) is a nonnegative integer that decreases by at least \(1\), so after at most \(e - 1\) iterations, \(\operatorname{ord}(b^2 a^{-1}) = 1\) and the algorithm terminates with \(b^2 a^{-1} \equiv 1 \pmod{p}\), i.e., \(b^2 \equiv a \pmod{p}\).
\end{proof}

In fact, if \(p \equiv 3 \pmod{4}\), then \(e = 1\), \(r = \frac{p-1}{2}\), and the initial candidate is \(b = a^{\frac{r+1}{2}} = a^{\frac{p+1}{4}}\).
Since \(b^2 a^{-1} \equiv a^{\frac{p-1}{2}} \equiv 1 \pmod{p}\), the \nameref{alg:modular-square-root} terminates immediately without executing the outer while loop, recovering the earlier closed-form solution for \(p \equiv 3 \pmod{4}\).

\begin{remark}[Complexity of the \nameref{alg:modular-square-root}]
    Initializations take \(O(\log^3 p)\) time.
    Since \(e = O(\log p)\), the outer while loop in line~\ref{alg:modular-square-root:outer-loop} executes at most \(O(\log p)\) times.
    In each iteration, the inner loop in line~\ref{alg:modular-square-root:inner-loop} and the update in line~\ref{alg:modular-square-root:update-b} use \(O(\log p)\) squarings and multiplications.
    Since each modular multiplication and squaring takes \(O(\log^2 p)\) time (as in \Cref{tab:arithmetic_runtimes}), the while loops perform a total of \(O(\log^2 p)\) modular operations, taking \(O(\log^4 p)\) time.
    Thus, the \nameref{alg:modular-square-root} is runs in \(O(\log^4 p)\) time.
\end{remark}
